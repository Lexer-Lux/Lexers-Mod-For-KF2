using System;
using System.Numerics;
using Lexer.Kf2.ForgetMeNot;

// Compile alongside GlintGeometry.cs in a disposable, source-only geometry probe.
// This does not launch a game and is not visual acceptance.
public static class GeometryChecks
{
    static void Require(bool okay, string message)
    { if (!okay) throw new Exception(message); }

    public static void Main()
    {
        int[] quad = { 0, 1, 2, 1, 3, 2 };
        Vector3[] door = { new(0,-2,5), new(2,-2,5), new(0,0,5), new(2,0,5) };
        Require(GlintGeometry.Surface(door, quad, new Vector3(1,-1,0), out var at), "door face should be found");
        Require(Vector3.Distance(at, new Vector3(1,-1,5)) < 0.001f, "hinge origin is not the door's visible surface centre");

        Vector3[] layered = {
            new(-1,-1,4),new(1,-1,4),new(-1,1,4),new(1,1,4),
            new(-1,-1,6),new(1,-1,6),new(-1,1,6),new(1,1,6) };
        int[] layers = { 0,1,2,1,3,2,4,5,6,5,7,6 };
        Require(GlintGeometry.Surface(layered, layers, Vector3.Zero, out at) && MathF.Abs(at.Z-4)<0.001f,
            "anchor must be on the nearest visible surface, not inside a model's box");
        int[] back = { 0,2,1,1,2,3 };
        Require(!GlintGeometry.Surface(door, back, new Vector3(1,-1,0), out _), "back-facing-only geometry has no visible anchor");

        Vector3[] frame = {
            new(-3,-3,5),new(-2,-3,5),new(-3,3,5),new(-2,3,5),
            new(2,-3,5),new(3,-3,5),new(2,3,5),new(3,3,5),
            new(-2,-3,5),new(2,-3,5),new(-2,-2,5),new(2,-2,5),
            new(-2,2,5),new(2,2,5),new(-2,3,5),new(2,3,5) };
        int[] bars = {0,1,2,1,3,2,4,5,6,5,7,6,8,9,10,9,11,10,12,13,14,13,15,14};
        Require(GlintGeometry.Surface(frame,bars,Vector3.Zero,out at), "hollow frame needs a surface anchor");
        Require(MathF.Abs(at.X)>=1.999f || MathF.Abs(at.Y)>=1.999f, "hollow centre must not receive a floating marker");
        Require(MathF.Abs(at.Z-5)<0.001f, "anchor remains on actual geometry");

        short[] turn = { 0,0,-4096, 0,4096,0, 4096,0,0 };
        Vector3 moved = GlintGeometry.Transform(new Vector3(10,-20,30),turn,new Vector3(100000,-1000,120000));
        Require(Vector3.Distance(moved,new Vector3(99970,-1020,120010))<0.001f,
            "full world coordinates and the submitted model rotation must survive");
        Require(!GlintGeometry.Surface(Array.Empty<Vector3>(),Array.Empty<int>(),Vector3.Zero,out _), "empty mesh must be rejected");
        Require(!GlintGeometry.Surface(new[]{Vector3.One,Vector3.One,Vector3.One},new[]{0,1,2},Vector3.Zero,out _),
            "degenerate geometry must be rejected");
        Vector3[] slope={new(-2,-2,2),new(2,-2,6),new(-2,2,2),new(2,2,6)};
        Require(GlintGeometry.Ray(slope,quad,Vector3.Zero,new Vector3(.5f,.5f,5),out at)
            &&MathF.Abs(at.Z-(4+at.X))<0.001f,"corner ray follows a sloped actual surface");
        Vector3 shift=new(150000,-10000,100000);
        for(int i=0;i<slope.Length;i++)slope[i]+=shift;
        Require(GlintGeometry.Ray(slope,quad,shift,shift+new Vector3(500,500,5000),out at)
            &&MathF.Abs((at.Z-shift.Z)-(4+at.X-shift.X))<.02f,"surface rays retain precision across the real map");
        Console.WriteLine("Forget Me Not geometry fixtures passed.");
    }
}
