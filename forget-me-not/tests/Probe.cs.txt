sealed class SilenceGuard:IDisposable
{
    static void Check()
    {
        if(!Runtime.Headless||!ConfigManager.Game.Muted||ConfigManager.Game.MasterVolume!=0)
            throw new Exception("Headless/mute guard failed");
        var type=typeof(Runtime).Assembly.GetType("RecompOne.Runtime.Host.Audio")!;
        foreach(string name in new[]{"_alc","_al","_mixerThread"})
            if(type.GetField(name,BindingFlags.Static|BindingFlags.NonPublic)!.GetValue(null)!=null)
                throw new Exception("Audio guard failed: "+name);
        if((uint)type.GetField("_source",BindingFlags.Static|BindingFlags.NonPublic)!.GetValue(null)! !=0)
            throw new Exception("Audio guard failed: source");
    }
    public SilenceGuard(){Check();typeof(Runtime).Assembly.GetType("RecompOne.Runtime.Host.Audio")!.GetMethod("Initialize")!.Invoke(null,null);Check();Console.WriteLine("PASS: audio remained device/context/mixer null, source zero");}
    public void Dispose(){Check();Console.WriteLine("PASS: headless/muted/master zero and audio uninitialized after checks");}
}
