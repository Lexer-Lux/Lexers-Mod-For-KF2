void AssertSilent()
{
    var audio = typeof(Runtime).Assembly.GetType("RecompOne.Runtime.Host.Audio")!;
    Require(new[] { "_alc", "_al", "_mixerThread" }.All(n => audio.GetField(n, BindingFlags.Static | BindingFlags.NonPublic)!.GetValue(null) == null)
            && (uint)audio.GetField("_source", BindingFlags.Static | BindingFlags.NonPublic)!.GetValue(null)! == 0,
        "no audio API, output source or mixer thread");
    Require(!RecompOne.Runtime.Host.GpuJobs.Claimed, "no graphics host thread started");
}

const uint First = 0x8016C544, Definition = 0x80172624, Colour = 0x8006E604;
const int Parent = 199, Child = 198;
uint Record(int slot) => First + (uint)slot * 0x7C;

void Require(bool condition, string name)
{
    if (!condition) throw new Exception("FAIL: " + name);
    Console.WriteLine("PASS: " + name);
}
void Near(float actual, float expected, string name) => Require(Math.Abs(actual - expected) < 0.00001f, name);
var type = typeof(HitFeedbackMod);
Array Flashes() => (Array)type.GetField("Flashes", BindingFlags.Static | BindingFlags.NonPublic)!.GetValue(null)!;
(long Start, long End) Times(int slot)
{
    object flash = Flashes().GetValue(slot)!;
    var t = flash.GetType();
    return ((long)t.GetProperty("Started")!.GetValue(flash)!, (long)t.GetProperty("Ends")!.GetValue(flash)!);
}
float Intensity(int slot, long time) => (float)type.GetMethod("Intensity", BindingFlags.Static | BindingFlags.NonPublic)!.Invoke(null, [slot, time])!;
void Settings(float seconds, float red)
{
    mod.OnUnload();
    Runtime.View.SetFloat("lexer.kf2.hitfeedback.maximumseconds", seconds);
    Runtime.View.SetFloat("lexer.kf2.hitfeedback.redintensity", red);
    mod.OnLoad();
}
void Hit(int before, int after, int target = Parent)
{
    memory.WriteU16(Record(Parent) + 0x1A, (ushort)before);
    cpu.A0 = (uint)target;
    HitFeedbackMod.BeforeDamage(cpu, memory);
    memory.WriteU16(Record(Parent) + 0x1A, (ushort)after);
    HitFeedbackMod.AfterDamage(cpu, memory);
}
void Submit(int slot)
{
    RetainedScene.BeginFrame(default(RetainedScene.View));
    ModelWalk.SetSubmit(ModelKind.Creature, slot, Record(slot));
    HitFeedbackMod.BeforeModel(cpu, memory);
    RetainedScene.Find(RetainedScene.Serial)!.Instances.Add(default);
    HitFeedbackMod.AfterModel(cpu, memory);
}
