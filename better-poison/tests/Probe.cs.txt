sealed class SilenceGuard:IDisposable
{
    static void Check()
    {
        var type=typeof(Runtime).Assembly.GetType("RecompOne.Runtime.Host.Audio")!;
        foreach(string name in new[]{"_alc","_al","_mixerThread"})
            if(type.GetField(name,BindingFlags.Static|BindingFlags.NonPublic)!.GetValue(null)!=null)
                throw new Exception("Audio silence preflight failed: "+name);
        if((uint)type.GetField("_source",BindingFlags.Static|BindingFlags.NonPublic)!.GetValue(null)! != 0)
            throw new Exception("Audio silence preflight failed: source");
    }
    public SilenceGuard(){Check();typeof(Runtime).Assembly.GetType("RecompOne.Runtime.Host.Audio")!.GetMethod("Initialize")!.Invoke(null,null);Check();Console.WriteLine("PASS: headless Audio.Initialize leaves device/context/mixer null and source zero");}
    public void Dispose(){Check();Console.WriteLine("PASS: audio remained uninitialized after check");}
}
